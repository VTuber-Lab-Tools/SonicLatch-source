cmake_minimum_required(VERSION 3.24)
project(soniclatch_wheel_diagnostic_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.11.1 EXACT REQUIRED COMPONENTS Core Widgets)
get_filename_component(PRODUCT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(QCP_DIR "${PRODUCT_DIR}/third_party/qcustomplot-2.1.1")
add_library(qcp_wheel_diagnostic STATIC "${QCP_DIR}/qcustomplot.cpp" "${QCP_DIR}/qcustomplot.h")
target_include_directories(qcp_wheel_diagnostic PUBLIC "${QCP_DIR}")
target_compile_definitions(qcp_wheel_diagnostic PUBLIC QT_NO_PRINTER)
target_link_libraries(qcp_wheel_diagnostic PUBLIC Qt6::Core Qt6::Widgets)
foreach(enabled 0 1)
 add_executable(wheel_diagnostic_${enabled} wheel_diagnostic_qt.cpp)
 target_include_directories(wheel_diagnostic_${enabled} PRIVATE "${PRODUCT_DIR}/src")
 target_compile_definitions(wheel_diagnostic_${enabled} PRIVATE SONICLATCH_ENABLE_WHEEL_DIAGNOSTIC=${enabled})
 target_link_libraries(wheel_diagnostic_${enabled} PRIVATE qcp_wheel_diagnostic)
 if(MSVC)
  target_compile_options(wheel_diagnostic_${enabled} PRIVATE /utf-8 /UNDEBUG)
 endif()
endforeach()
