cmake_minimum_required(VERSION 3.24)
project(pitatuner_normal_source_dock_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.11.1 EXACT REQUIRED COMPONENTS Core Widgets)
get_filename_component(PRODUCT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(QCP_SOURCE_DIR "${PRODUCT_DIR}/third_party/qcustomplot-2.1.1")
set(OBS_HEADERS "" CACHE PATH "Verified target libobs headers; OBS 32.2.2 recommended")
set(OBS_CONFIG_DIR "" CACHE PATH "Verified generated obsconfig.h directory")
add_library(qcp STATIC "${QCP_SOURCE_DIR}/qcustomplot.cpp" "${QCP_SOURCE_DIR}/qcustomplot.h")
target_include_directories(qcp PUBLIC "${QCP_SOURCE_DIR}")
target_compile_definitions(qcp PUBLIC QT_NO_PRINTER)
target_link_libraries(qcp PUBLIC Qt6::Core Qt6::Widgets)
add_library(pitatuner_normal_source_dock MODULE normal-source-dock-module.cpp "${PRODUCT_DIR}/src/display_adapter.cpp" "${PRODUCT_DIR}/integrations/waveform-model/waveform-model-draft/waveform_model.cpp")
target_include_directories(pitatuner_normal_source_dock PRIVATE "${PRODUCT_DIR}/src" "${OBS_HEADERS}" "${OBS_HEADERS}/../frontend/api" "${OBS_CONFIG_DIR}")
target_link_libraries(pitatuner_normal_source_dock PRIVATE qcp)
# Public binary name only; CMake target and persistent OBS IDs stay stable.
set_target_properties(pitatuner_normal_source_dock PROPERTIES OUTPUT_NAME "SonicLatch")

target_compile_definitions(pitatuner_normal_source_dock PRIVATE SONICLATCH_ENABLE_EMBEDDED_QA=0)
# Development observation only; public builds keep this OFF.
option(SONICLATCH_BUILD_WHEEL_DIAGNOSTIC "Compile bounded, opt-in wheel diagnostics" OFF)
target_compile_definitions(pitatuner_normal_source_dock PRIVATE SONICLATCH_ENABLE_WHEEL_DIAGNOSTIC=$<BOOL:${SONICLATCH_BUILD_WHEEL_DIAGNOSTIC}>)
option(SONICLATCH_BUILD_QA "Build isolated fixtures and development tests" OFF)
if(SONICLATCH_BUILD_QA)
 include("${PRODUCT_DIR}/qa/development-targets.cmake")
endif()
