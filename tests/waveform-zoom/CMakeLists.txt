cmake_minimum_required(VERSION 3.24)
project(soniclatch_waveform_zoom_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.11.1 EXACT REQUIRED COMPONENTS Core Widgets)
get_filename_component(PRODUCT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(QCP_DIR "${PRODUCT_DIR}/third_party/qcustomplot-2.1.1")
add_library(qcp_zoom_test STATIC "${QCP_DIR}/qcustomplot.cpp" "${QCP_DIR}/qcustomplot.h")
target_include_directories(qcp_zoom_test PUBLIC "${QCP_DIR}")
target_compile_definitions(qcp_zoom_test PUBLIC QT_NO_PRINTER)
target_link_libraries(qcp_zoom_test PUBLIC Qt6::Core Qt6::Widgets)
add_executable(waveform_zoom_qt "${PRODUCT_DIR}/tests/waveform_zoom_qt.cpp")
target_include_directories(waveform_zoom_qt PRIVATE "${PRODUCT_DIR}/src")
target_link_libraries(waveform_zoom_qt PRIVATE qcp_zoom_test)
add_executable(waveform_viewport_tests "${PRODUCT_DIR}/tests/waveform_viewport_tests.cpp")
target_include_directories(waveform_viewport_tests PRIVATE "${PRODUCT_DIR}/src")
