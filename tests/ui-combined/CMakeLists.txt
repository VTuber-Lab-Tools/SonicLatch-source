cmake_minimum_required(VERSION 3.24)
project(soniclatch_combined_ui_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.11.1 EXACT REQUIRED COMPONENTS Core Widgets)
get_filename_component(PRODUCT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(QCP_DIR "${PRODUCT_DIR}/third_party/qcustomplot-2.1.1")
add_library(qcp_combined_ui STATIC "${QCP_DIR}/qcustomplot.cpp" "${QCP_DIR}/qcustomplot.h")
target_include_directories(qcp_combined_ui PUBLIC "${QCP_DIR}")
target_compile_definitions(qcp_combined_ui PUBLIC QT_NO_PRINTER)
target_link_libraries(qcp_combined_ui PUBLIC Qt6::Core Qt6::Widgets)
foreach(target ui_combined_qt_tests source_batch_qt_tests waveform_zoom_qt simple_settings_ui_tests manual_edit_commit_tests first_auto_guard_qt_tests correction_range_qt_tests)
 add_executable(${target} "${PRODUCT_DIR}/tests/${target}.cpp")
 target_include_directories(${target} PRIVATE "${PRODUCT_DIR}/src")
 target_link_libraries(${target} PRIVATE qcp_combined_ui)
 if(MSVC)
  target_compile_options(${target} PRIVATE /utf-8 /UNDEBUG)
 endif()
endforeach()
add_executable(ui_combined_pure_tests "${PRODUCT_DIR}/tests/ui_combined_pure_tests.cpp")
target_include_directories(ui_combined_pure_tests PRIVATE "${PRODUCT_DIR}/src")

add_executable(display_offset_history_tests "${PRODUCT_DIR}/tests/display_offset_history_tests.cpp")
target_include_directories(display_offset_history_tests PRIVATE "${PRODUCT_DIR}/src")
